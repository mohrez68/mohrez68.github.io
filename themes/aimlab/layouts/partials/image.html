{{- /* Usage: partial "image.html" (dict "resource" $res "width" 1200 "format" "webp" "quality" 82) partial "image.html" (dict "resource" $res "width" 800 "height" 800 "mode" "fit" "format" "webp") Returns a dict with src, width, height. mode "fit" scales the image down to fit within width x height without upscaling. */ -}} {{- $r := .resource -}} {{- $width := .width | default 1200 -}} {{- $height := .height | default $width -}} {{- $mode := .mode | default "resize" -}} {{- $format := .format | default "webp" -}} {{- $quality := .quality | default 82 -}} {{- if and $r (eq $r.ResourceType "image") (ne $r.MediaType.SubType "svg") -}} {{- if eq $mode "fit" -}} {{- $w := cond (gt $r.Width $width) $width $r.Width -}} {{- $h := cond (gt $r.Height $height) $height $r.Height -}} {{- $r = $r.Fit (printf "%dx%d %s q%d" $w $h $format $quality) -}} {{- else -}} {{- $r = $r.Resize (printf "%dx %s q%d" $width $format $quality) -}} {{- end -}} {{- end -}} {{- return (dict "src" $r.RelPermalink "width" $r.Width "height" $r.Height) -}}
